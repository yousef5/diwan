# Diwan 1.0 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build Diwan 1.0, a professional multi-account WhatsApp Web desktop client for Linux, per
`docs/superpowers/specs/2026-10-22-diwan-1.0-design.md`.

**Architecture:** Electron app with three processes. The **main process** owns all state and logic:
the store, the account registry, guest hardening, notifications, the tray and the importer.
The **shell renderer** (React) only draws a `Snapshot` pushed by main and sends intents. A
**guest preload** inside every web.whatsapp.com `<webview>` reports events (notifications,
online/offline, avatar). Unread counts come from the guest page title, read in main. Pure logic
lives in small modules with unit tests. Electron glue is thin and covered by an E2E smoke test and
a manual checklist.

**Tech Stack:** Electron 44.9.3 · electron-vite 5 · Vite 7 · React 19 · TypeScript 5.9 · Tailwind
CSS 4 · zod 4 · zustand 5 · Vitest 4 (+ jsdom) · Playwright 1.64 (`_electron`) · Biome 2 ·
electron-builder 26 · Bun (package manager).

**Conventions used everywhere:** 2-space indent, single quotes, no semicolons, named exports,
`import type` for types. Pure modules never import `electron`. Every commit message ends with
the two attribution lines:

```ts
export const WA_ORIGIN = 'https://web.whatsapp.com'

export type LinkAction =
  | { kind: 'internal' }
  | { kind: 'chat'; url: string }
  | { kind: 'external'; url: string }
  | { kind: 'download'; url: string }
  | { kind: 'blocked'; reason: string }

// A chat link only opens the chat (with optional draft text). Nothing is ever sent.
export function chatUrl(phone: string | null, text: string | null): string {
  const params = new URLSearchParams()
  const digits = phone?.replace(/\D/g, '')
  if (digits) params.set('phone', digits)
  if (text) params.set('text', text)
  const query = params.toString()
  return `${WA_ORIGIN}/send${query ? `?${query}` : ''}`
}

export function classifyLink(raw: string): LinkAction {
  let u: URL
  try {
    u = new URL(raw)
  } catch {
    return { kind: 'blocked', reason: 'invalid url' }
  }

  switch (u.protocol) {
    case 'whatsapp:':
      if (u.host === 'send' || u.pathname.replace(/^\/+/, '') === 'send') {
        return { kind: 'chat', url: chatUrl(u.searchParams.get('phone'), u.searchParams.get('text')) }
      }
      return { kind: 'blocked', reason: 'unsupported whatsapp: link' }
    case 'blob:':
      return u.origin === WA_ORIGIN ? { kind: 'download', url: raw } : { kind: 'blocked', reason: 'foreign blob' }
    case 'mailto:':
    case 'tel:':
      return { kind: 'external', url: raw }
    case 'http:':
    case 'https:':
      break
    default:
      return { kind: 'blocked', reason: `scheme ${u.protocol}` }
  }

  const host = u.hostname.toLowerCase()
  if (u.protocol === 'https:' && host === 'web.whatsapp.com') {
    if (u.pathname === '/send' || u.pathname.startsWith('/send/')) {
      return { kind: 'chat', url: chatUrl(u.searchParams.get('phone'), u.searchParams.get('text')) }
    }
    return { kind: 'internal' }
  }
  if (host === 'wa.me' || host === 'www.wa.me') {
    const m = u.pathname.match(/^\/(\+?\d{6,15})\/?$/)
    if (m) return { kind: 'chat', url: chatUrl(m[1], u.searchParams.get('text')) }
    return { kind: 'external', url: u.href }
  }
  const isWaHost = host === 'api.whatsapp.com' || host === 'whatsapp.com' || host === 'www.whatsapp.com'
  if (isWaHost && u.pathname.replace(/\/$/, '') === '/send') {
    return { kind: 'chat', url: chatUrl(u.searchParams.get('phone'), u.searchParams.get('text')) }
  }
  return { kind: 'external', url: u.href }
}
```

```ts
import type { Account, AccountRuntime } from './types'

// Arabic-Indic (U+0660–0669) and Extended Arabic-Indic (U+06F0–06F9) digits → ASCII.
const NON_ASCII_DIGITS = /[٠-٩۰-۹]/g

export function toAsciiDigits(s: string): string {
  return s.replace(NON_ASCII_DIGITS, (d) => String(d.charCodeAt(0) & 0xf))
}

// WhatsApp Web puts the unread count in the page title: "(3) WhatsApp", "(٣) واتساب".
export function parseUnread(title: string): number {
  const m = toAsciiDigits(title).match(/\((\d{1,5})\)/)
  return m ? Number(m[1]) : 0
}

export function unreadTotal(accounts: Account[], runtime: Record<string, AccountRuntime>): number {
  let total = 0
  for (const a of accounts) if (!a.muted) total += runtime[a.id]?.unread ?? 0
  return total
}

export const BADGE_KEYS = ['none', '1', '2', '3', '4', '5', '6', '7', '8', '9', '9+'] as const
export type BadgeKey = (typeof BADGE_KEYS)[number]

export function badgeKey(total: number): BadgeKey {
  if (total <= 0) return 'none'
  if (total > 9) return '9+'
  return String(total) as BadgeKey
}

export function windowTitle(total: number): string {
  return total > 0 ? `Diwan (${total})` : 'Diwan'
}
```

```ts
// Matches on `code` (physical key), not `key`, so shortcuts work with the Arabic layout active.
export interface KeyInput {
  type: string
  code: string
  key: string
  control: boolean
  meta: boolean
  shift: boolean
  alt: boolean
}

export type ShortcutAction =
  | { type: 'account'; index: number }
  | { type: 'next' }
  | { type: 'prev' }
  | { type: 'reload' }
  | { type: 'remount' }
  | { type: 'zoom'; delta: 1 | -1 | 0 }
  | { type: 'settings' }
  | { type: 'hide' }
  | { type: 'quit' }

export function matchShortcut(i: KeyInput): ShortcutAction | null {
  if (i.type !== 'keyDown' || i.alt) return null
  if (i.code === 'F5' && !i.control && !i.meta) return { type: 'reload' }
  if (!i.control && !i.meta) return null

  const digit = /^(?:Digit|Numpad)([1-9])$/.exec(i.code)
  if (digit) return i.shift ? null : { type: 'account', index: Number(digit[1]) - 1 }

  switch (i.code) {
    case 'Tab':
      return i.shift ? { type: 'prev' } : { type: 'next' }
    case 'PageDown':
      return i.shift ? null : { type: 'next' }
    case 'PageUp':
      return i.shift ? null : { type: 'prev' }
    case 'KeyR':
      return i.shift ? { type: 'remount' } : { type: 'reload' }
    case 'Equal':
    case 'NumpadAdd':
      return { type: 'zoom', delta: 1 }
    case 'Minus':
    case 'NumpadSubtract':
      return i.shift ? null : { type: 'zoom', delta: -1 }
    case 'Digit0':
    case 'Numpad0':
      return i.shift ? null : { type: 'zoom', delta: 0 }
    case 'Comma':
      return i.shift ? null : { type: 'settings' }
    case 'KeyW':
      return i.shift ? null : { type: 'hide' }
    case 'KeyQ':
      return i.shift ? null : { type: 'quit' }
    default:
      return null
  }
}
```

```ts
  importedFromNexus: z.boolean().default(false),
  accounts: z.array(AccountSchema).default(() => []),
  settings: SettingsSchema.default(() => clone(DEFAULT_SETTINGS)),
})
```

```ts
import { SCHEMA_VERSION } from './schema'

export type Migration = (raw: Record<string, unknown>) => Record<string, unknown>

// Key = the version a step migrates FROM. Add `1: (x) => ({ ...x, newField: … })` when v2 exists.
export const MIGRATIONS: Record<number, Migration> = {}

export class NewerSchemaError extends Error {}

export function migrate(raw: unknown, migrations: Record<number, Migration> = MIGRATIONS, target = SCHEMA_VERSION): unknown {
  if (typeof raw !== 'object' || raw === null || Array.isArray(raw)) return raw
  let obj: Record<string, unknown> = { ...(raw as Record<string, unknown>) }
  let version = typeof obj.schemaVersion === 'number' ? obj.schemaVersion : 1
  if (version > target) throw new NewerSchemaError(`settings file is schema ${version}, this Diwan understands ${target}`)
  while (version < target) {
    const step = migrations[version]
    if (!step) throw new Error(`no migration from ${version}`)
    obj = step(obj)
    version++
  }
  obj.schemaVersion = version
  return obj
}
```

```ts
  private data: DiwanFile
  private timer: ReturnType<typeof setTimeout> | null = null
  private mainGood = false
  private _recovery = false
  private _source: LoadSource = 'default'

  constructor(dir: string, private readonly debounceMs = 300) {
    mkdirSync(dir, { recursive: true })
    this.path = join(dir, 'diwan.json')
    this.backupPath = `${this.path}.bak`
    this.data = this.load()
  }

  get recovery(): boolean {
    return this._recovery
  }

  get source(): LoadSource {
    return this._source
  }

  get(): DiwanFile {
    return this.data
  }

  update(mutator: (draft: DiwanFile) => void): void {
    if (this._recovery) throw new Error('settings are in recovery mode and cannot be changed')
    const draft = structuredClone(this.data)
    mutator(draft)
    this.data = parseDiwanFile(draft)
    this.schedule()
  }

  flush(): void {
    if (!this.timer) return
    clearTimeout(this.timer)
    this.timer = null
    this.writeNow()
  }

  startFresh(): void {
    const stamp = Date.now()
    if (existsSync(this.path)) renameSync(this.path, `${this.path}.broken-${stamp}`)
    if (existsSync(this.backupPath)) renameSync(this.backupPath, `${this.backupPath}.broken-${stamp}`)
    this._recovery = false
    this.mainGood = false
    this.data = defaultFile()
    this.writeNow()
  }

  private load(): DiwanFile {
    // A file written by a newer Diwan is never read through a backup or overwritten (no downgrade).
    if (this.isNewer(this.path) || this.isNewer(this.backupPath)) {
      this._recovery = true
      this._source = 'default'
      return defaultFile()
    }
    const main = existsSync(this.path) ? this.readParsed(this.path) : undefined
    if (main) {
      this._source = 'file'
      this.mainGood = true
      return main
    }
    const backup = existsSync(this.backupPath) ? this.readParsed(this.backupPath) : undefined
    if (backup) {
      this._source = 'backup'
      return backup
    }
    this._source = 'default'
    if (main === null || backup === null) this._recovery = true
    return defaultFile()
  }

  private isNewer(p: string): boolean {
    if (!existsSync(p)) return false
    try {
      migrate(JSON.parse(readFileSync(p, 'utf8')))
      return false
    } catch (e) {
      return e instanceof NewerSchemaError
    }
  }

  // undefined = absent (handled by caller), null = present but unusable
  private readParsed(p: string): DiwanFile | null {
    try {
      return parseDiwanFile(migrate(JSON.parse(readFileSync(p, 'utf8'))))
    } catch {
      return null
    }
  }

  private schedule(): void {
    if (this.debounceMs === 0) {
      this.writeNow()
      return
    }
    if (this.timer) clearTimeout(this.timer)
    this.timer = setTimeout(() => {
      this.timer = null
      this.writeNow()
    }, this.debounceMs)
  }

  private writeNow(): void {
    if (this._recovery) return
    const tmp = `${this.path}.tmp`
    const fd = openSync(tmp, 'w')
    try {
      writeSync(fd, `${JSON.stringify(this.data, null, 2)}\n`)
      fsyncSync(fd)
    } finally {
      closeSync(fd)
    }
    if (this.mainGood) copyFileSync(this.path, this.backupPath)
    renameSync(tmp, this.path)
    this.mainGood = true
  }
}
```

```ts
import { join } from 'node:path'
import { z } from 'zod'
import { colorForIndex } from '../../shared/palette'
import type { ImportItem, ImportPlan } from '../../shared/types'

export interface PlanDeps {
  readText(path: string): string | null
  listDir(path: string): string[]
  dirSize(path: string): number
}

const NexusConfig = z.object({
  accounts: z.array(z.object({ id: z.string().min(1), name: z.string(), order: z.number().optional() })),
})

const unavailable = (reason: string): ImportPlan => ({ available: false, reason, items: [], orphanDirs: 0, nexusRunning: false })

export function planImport(paths: { nexusConfig: string; nexusPartitions: string }, deps: PlanDeps): ImportPlan {
  const raw = deps.readText(paths.nexusConfig)
  if (raw === null) return unavailable('Nexus is not installed on this computer')
  let parsed: z.infer<typeof NexusConfig>
  try {
    parsed = NexusConfig.parse(JSON.parse(raw))
  } catch {
    return unavailable('Nexus settings file is damaged')
  }

  const dirs = new Set(deps.listDir(paths.nexusPartitions))
  const accounts = [...parsed.accounts].sort((a, b) => (a.order ?? 0) - (b.order ?? 0))
  const items: ImportItem[] = accounts
    .filter((a) => dirs.has(`wa-${a.id}`))
    .map((a, i) => {
      const sourceDir = join(paths.nexusPartitions, `wa-${a.id}`)
      return {
        sourceId: a.id,
        name: a.name.trim().slice(0, 40) || `Account ${i + 1}`,
        color: colorForIndex(i),
        sourceDir,
        bytes: deps.dirSize(sourceDir),
      }
    })
  const live = new Set(accounts.map((a) => `wa-${a.id}`))
  const orphanDirs = [...dirs].filter((d) => d.startsWith('wa-') && !live.has(d)).length
  if (items.length === 0) return { ...unavailable('Nexus has no signed-in accounts'), orphanDirs }
  return { available: true, reason: null, items, orphanDirs, nexusRunning: false }
}
```

```ts
import { join } from 'node:path'

export function sanitizeFilename(name: string): string {
  const s = name.replace(/[/\\\0]/g, '_').replace(/^\.+/, '').trim().slice(0, 200)
  return s || 'download'
}

export function uniquePath(dir: string, filename: string, exists: (p: string) => boolean): string {
  const clean = sanitizeFilename(filename)
  const dot = clean.lastIndexOf('.')
  const base = dot > 0 ? clean.slice(0, dot) : clean
  const ext = dot > 0 ? clean.slice(dot) : ''
  let candidate = join(dir, clean)
  for (let n = 1; exists(candidate); n++) candidate = join(dir, `${base} (${n})${ext}`)
  return candidate
}
```

```ts
import type { Rect } from '../shared/types'

const MIN_VISIBLE = 100

function overlap(a: Rect, b: Rect): { w: number; h: number } {
  const w = Math.min(a.x + a.width, b.x + b.width) - Math.max(a.x, b.x)
  const h = Math.min(a.y + a.height, b.y + b.height) - Math.max(a.y, b.y)
  return { w: Math.max(0, w), h: Math.max(0, h) }
}

// Saved bounds are reused only if at least 100×100 px of the window lands on a current display.
export function visibleBounds(
  saved: Rect | null,
  displays: Rect[],
  fallback: { width: number; height: number },
): Rect | { width: number; height: number } {
  if (!saved) return fallback
  const onScreen = displays.some((d) => {
    const o = overlap(saved, d)
    return o.w >= MIN_VISIBLE && o.h >= MIN_VISIBLE
  })
  return onScreen ? saved : { width: saved.width, height: saved.height }
}
```

```ts
export interface MenuParams {
  x: number
  y: number
  isEditable: boolean
  selectionText: string
  misspelledWord: string
  dictionarySuggestions: string[]
  linkURL: string
  mediaType: string
  srcURL: string
  editFlags: { canCut: boolean; canCopy: boolean; canPaste: boolean; canSelectAll: boolean }
}

export type EditRole = 'cut' | 'copy' | 'paste' | 'pasteAndMatchStyle' | 'selectAll'

export type MenuEntry =
  | { type: 'separator' }
  | { type: 'item'; id: string; label: string; enabled?: boolean }
  | { type: 'role'; role: EditRole; label?: string; enabled?: boolean }

const shorten = (s: string, n: number) => (s.length > n ? `${s.slice(0, n)}…` : s)

export function buildMenu(p: MenuParams, o: { dev: boolean }): MenuEntry[] {
  const groups: MenuEntry[][] = []

  if (p.misspelledWord) {
    const g: MenuEntry[] = p.dictionarySuggestions
      .slice(0, 5)
      .map((s): MenuEntry => ({ type: 'item', id: `spell:${s}`, label: s }))
    if (g.length === 0) g.push({ type: 'item', id: 'noop', label: 'No suggestions', enabled: false })
    g.push({ type: 'item', id: 'addWord', label: `Add “${p.misspelledWord}” to dictionary` })
    groups.push(g)
  }

  if (p.linkURL && !p.linkURL.startsWith('blob:')) {
    groups.push([
      { type: 'item', id: 'openLink', label: 'Open link in browser' },
      { type: 'item', id: 'copyLink', label: 'Copy link' },
    ])
  }

  if (p.mediaType === 'image' && p.srcURL) {
    groups.push([
      { type: 'item', id: 'copyImage', label: 'Copy image' },
      { type: 'item', id: 'saveImage', label: 'Save image to Downloads' },
    ])
  }

  if (p.isEditable) {
    groups.push([
      { type: 'role', role: 'cut', enabled: p.editFlags.canCut },
      { type: 'role', role: 'copy', enabled: p.editFlags.canCopy },
      { type: 'role', role: 'paste', enabled: p.editFlags.canPaste },
      { type: 'role', role: 'pasteAndMatchStyle', label: 'Paste as plain text', enabled: p.editFlags.canPaste },
      { type: 'role', role: 'selectAll', enabled: p.editFlags.canSelectAll },
    ])
  } else if (p.selectionText.trim()) {
    groups.push([
      { type: 'role', role: 'copy' },
      { type: 'item', id: 'search', label: `Search the web for “${shorten(p.selectionText.trim(), 24)}”` },
    ])
  }

  if (o.dev && groups.length) groups.push([{ type: 'item', id: 'inspect', label: 'Inspect element' }])

  return groups.flatMap((g, i) => (i === 0 ? g : [{ type: 'separator' } as MenuEntry, ...g]))
}
```

```ts
import { appendFileSync, existsSync, mkdirSync, renameSync, rmSync, statSync } from 'node:fs'
import { join } from 'node:path'

export interface Logger {
  info(msg: string, extra?: unknown): void
  warn(msg: string, extra?: unknown): void
  error(msg: string, extra?: unknown): void
}

const KEEP = 3

function describe(extra: unknown): string {
  if (extra === undefined) return ''
  if (extra instanceof Error) return ` ${extra.name}: ${extra.message}`
  try {
    return ` ${JSON.stringify(extra)}`
  } catch {
    return ' [unserialisable]'
  }
}

// Plain rotating file log in <userData>/logs. Callers never pass message text, only metadata.
export function createLogger(dir: string, opts: { maxBytes?: number; echo?: boolean } = {}): Logger {
  const maxBytes = opts.maxBytes ?? 1_000_000
  const echo = opts.echo ?? true
  const file = join(dir, 'main.log')
  try {
    mkdirSync(dir, { recursive: true })
  } catch {}

  const rotate = () => {
    try {
      if (statSync(file).size < maxBytes) return
    } catch {
      return
    }
    rmSync(`${file}.${KEEP}`, { force: true })
    for (let i = KEEP - 1; i >= 1; i--) if (existsSync(`${file}.${i}`)) renameSync(`${file}.${i}`, `${file}.${i + 1}`)
    renameSync(file, `${file}.1`)
  }

  const write = (level: 'info' | 'warn' | 'error', msg: string, extra?: unknown) => {
    const line = `${new Date().toISOString()} ${level} ${msg}${describe(extra)}\n`
    try {
      rotate()
      appendFileSync(file, line)
    } catch {}
    if (echo) (level === 'error' ? console.error : console.log)(line.trimEnd())
  }

  return {
    info: (m, e) => write('info', m, e),
    warn: (m, e) => write('warn', m, e),
    error: (m, e) => write('error', m, e),
  }
}
```

```ts
import { randomUUID } from 'node:crypto'
import { EventEmitter } from 'node:events'
import type { Account, AccountRuntime, DiwanFile, Settings, SettingsPatch, Snapshot, WindowState } from '../shared/types'
import type { Store } from './store/store'

const DEFAULT_RUNTIME: AccountRuntime = { state: 'loading', unread: 0, avatar: null }

// Owns everything the shell renders. Persistent data lives in the Store; runtime (unread,
// load state, avatar) and the active account live here. Emits one 'changed' per tick.
export class StateHub extends EventEmitter<{ changed: [Snapshot] }> {
  importAvailable = false
  trayAvailable = true
  private runtime = new Map<string, AccountRuntime>()
  private activeId: string | null
  private pending = false

  constructor(readonly store: Store) {
    super()
    this.activeId = this.accounts()[0]?.id ?? null
  }

  accounts(): Account[] {
    return this.store.get().accounts
  }

  account(id: string): Account | undefined {
    return this.accounts().find((a) => a.id === id)
  }

  settings(): Settings {
    return this.store.get().settings
  }

  active(): string | null {
    return this.activeId
  }

  runtimeOf(id: string): AccountRuntime {
    return this.runtime.get(id) ?? DEFAULT_RUNTIME
  }

  snapshot(): Snapshot {
    const accounts = this.accounts()
    return {
      accounts,
      settings: this.settings(),
      runtime: Object.fromEntries(accounts.map((a) => [a.id, this.runtimeOf(a.id)])),
      activeId: this.activeId,
      recovery: this.store.recovery,
      importAvailable: this.importAvailable,
      trayAvailable: this.trayAvailable,
    }
  }

  notify(): void {
    if (this.pending) return
    this.pending = true
    queueMicrotask(() => {
      this.pending = false
      this.emit('changed', this.snapshot())
    })
  }

  addAccount(name: string, color: string, id: string = randomUUID()): Account {
    const account: Account = {
      id,
      name: name.trim(),
      color,
      order: this.accounts().length,
      zoom: 0,
      muted: false,
      customAvatar: false,
      createdAt: new Date().toISOString(),
    }
    this.store.update((f) => {
      f.accounts.push(account)
    })
    if (!this.activeId) this.activeId = id
    this.persistNow()
    this.notify()
    return account
  }

  updateAccount(id: string, patch: Partial<Pick<Account, 'name' | 'color' | 'muted' | 'zoom' | 'customAvatar'>>): void {
    this.mutate((f) => {
      const a = f.accounts.find((x) => x.id === id)
      if (a) Object.assign(a, patch)
    })
  }

  removeAccount(id: string): void {
    this.mutate((f) => {
      f.accounts = f.accounts.filter((a) => a.id !== id).map((a, i) => ({ ...a, order: i }))
    })
    this.runtime.delete(id)
    if (this.activeId === id) this.activeId = this.accounts()[0]?.id ?? null
    this.notify()
    this.persistNow()
  }

  reorder(ids: string[]): void {
    const pos = new Map(ids.map((id, i) => [id, i]))
    this.mutate((f) => {
      f.accounts.sort((a, b) => (pos.get(a.id) ?? 1e9) - (pos.get(b.id) ?? 1e9) || a.order - b.order)
      f.accounts.forEach((a, i) => {
        a.order = i
      })
    })
  }

  setActive(id: string): boolean {
    if (!this.account(id) || this.activeId === id) return false
    this.activeId = id
    this.notify()
    return true
  }

  cycle(dir: 1 | -1): string | null {
    const list = this.accounts()
    if (list.length === 0) return null
    const i = list.findIndex((a) => a.id === this.activeId)
    const next = list[(i + dir + list.length) % list.length]
    this.setActive(next.id)
    return next.id
  }

  setRuntime(id: string, patch: Partial<AccountRuntime>): void {
    const cur = this.runtimeOf(id)
    const next = { ...cur, ...patch }
    if (cur.state === next.state && cur.unread === next.unread && cur.avatar === next.avatar) return
    this.runtime.set(id, next)
    this.notify()
  }

  updateSettings(patch: SettingsPatch): void {
    this.mutate((f) => {
      const { notifications, ...rest } = patch
      Object.assign(f.settings, rest)
      if (notifications) Object.assign(f.settings.notifications, notifications)
    })
  }

  // Window geometry changes constantly; save it without pushing a snapshot to the shell.
  persistWindow(w: WindowState): void {
    if (this.store.recovery) return
    this.store.update((f) => {
      f.settings.window = w
    })
  }

  markImported(): void {
    this.mutate((f) => {
      f.importedFromNexus = true
    })
    this.importAvailable = false
    this.notify()
    this.persistNow()
  }

  // Account changes are written at once, not debounced. A failed write is already reported
  // (onError) and retried by the store with backoff, so it must not abort the state update.
  private persistNow(): void {
    if (this.store.recovery) return
    try {
      this.store.flush()
    } catch {}
  }

  private mutate(fn: (f: DiwanFile) => void): void {
    if (this.store.recovery) return
    this.store.update(fn)
    this.notify()
  }
}
```

```ts
// this; messages from unknown senders are dropped.
export class Registry {
  private byWc = new Map<number, string>()
  private byAccount = new Map<string, WebContents>()

  attach(wc: WebContents, accountId: string): void {
    this.byWc.set(wc.id, accountId)
    this.byAccount.set(accountId, wc)
  }

  detach(wcId: number): void {
    const accountId = this.byWc.get(wcId)
    if (accountId === undefined) return
    this.byWc.delete(wcId)
    if (this.byAccount.get(accountId)?.id === wcId) this.byAccount.delete(accountId)
  }

    return this.byWc.get(wcId)

    const wc = this.byAccount.get(accountId)
    return wc && !wc.isDestroyed() ? wc : undefined
  }
```

```ts
export interface HangSource {
  on(event: 'unresponsive' | 'responsive' | 'destroyed', listener: () => void): unknown
}

// Chromium reports 'unresponsive' after a few seconds of a frozen page. Give it 10 s more,
// then ask for a remount. Never forcefullyCrashRenderer a webview guest: that takes the main
// process down ~30 s later (seen in Nexus).
export function watchHang(wc: HangSource, onHung: () => void, graceMs = 10_000): void {
  let timer: ReturnType<typeof setTimeout> | null = null
  const clear = () => {
    if (timer) clearTimeout(timer)
    timer = null
  }
  wc.on('unresponsive', () => {
    if (timer) return
    timer = setTimeout(() => {
      timer = null
      onHung()
    }, graceMs)
  })
  wc.on('responsive', clear)
  wc.on('destroyed', clear)
}
```

```ts
import { randomUUID } from 'node:crypto'
import { constants, existsSync, readdirSync, readFileSync, readlinkSync, statSync } from 'node:fs'
import { cp, rm } from 'node:fs/promises'
import { join } from 'node:path'
import type { ImportPlan, ImportResult } from '../../shared/types'
import type { Logger } from '../log'
import type { StateHub } from '../state'
import type { PlanDeps } from './plan'

export function dirSize(path: string): number {
  let total = 0
  const walk = (dir: string) => {
    let entries: import('node:fs').Dirent[]
    try {
      entries = readdirSync(dir, { withFileTypes: true })
    } catch {
      return
    }
    for (const e of entries) {
      const full = join(dir, e.name)
      if (e.isDirectory()) walk(full)
      else if (e.isFile()) {
        try {
          total += statSync(full).size
        } catch {}
      }
    }
  }
  walk(path)
  return total
}

export const fsPlanDeps: PlanDeps = {
  readText: (p) => {
    try {
      return readFileSync(p, 'utf8')
    } catch {
      return null
    }
  },
  listDir: (p) => {
    try {
      return readdirSync(p)
    } catch {
      return []
    }
  },
  dirSize,
}

// Chromium keeps <userData>/SingletonLock → "<hostname>-<pid>" while the app runs.
export function nexusRunning(nexusUserData: string, pidAlive = (pid: number) => existsSync(`/proc/${pid}`)): boolean {
  try {
    const target = readlinkSync(join(nexusUserData, 'SingletonLock'))
    const pid = Number(target.split('-').pop())
    return Number.isInteger(pid) && pid > 0 && pidAlive(pid)
  } catch {
    return false
  }
}

// Copies each session folder (reflink on btrfs, so instant) into a new partition, then
// registers the account. Nexus's files are only read.
export async function applyImport(
  plan: ImportPlan,
  d: { hub: StateHub; partitionsDir: string; setupSession(id: string): void; log: Logger },
): Promise<ImportResult> {
  const result: ImportResult = { imported: [], failed: [] }
  for (const item of plan.items) {
    const id = randomUUID()
    const dest = join(d.partitionsDir, `acc-${id}`)
    try {
      await cp(item.sourceDir, dest, { recursive: true, errorOnExist: true, force: false, mode: constants.COPYFILE_FICLONE })
      d.setupSession(id)
      d.hub.addAccount(item.name, item.color, id)
      result.imported.push(item.name)
      d.log.info('imported account from Nexus', { id })
    } catch (e) {
      await rm(dest, { recursive: true, force: true })
      result.failed.push({ name: item.name, error: e instanceof Error ? e.message : String(e) })
      d.log.error('import failed', e)
    }
  }
  if (result.imported.length > 0) d.hub.markImported()
  return result
}
```

```ts
const pendingDownloads = new Set<string>()

function lstatExists(p: string): boolean {
  try {
    lstatSync(p)
    return true
  } catch {
    return false
  }
}

// Must run before the account's <webview> attaches (startup, add, import).
export function setupGuestSession(accountId: string, d: { downloads: string; log: Logger }): Session {
  const ses = session.fromPartition(partitionOf(accountId))
  if (configured.has(ses)) return ses
  configured.add(ses)

  ses.setPermissionRequestHandler((_wc, permission, callback, details) =>
    callback(allowGuestPermission(permission, details.requestingUrl)),
  )
  ses.setPermissionCheckHandler((_wc, permission, requestingOrigin) => allowGuestPermission(permission, requestingOrigin))
  ses.setUserAgent(chromeUserAgent())
  try {
    ses.setSpellCheckerLanguages(pickSpellLanguages(ses.availableSpellCheckerLanguages))
  } catch (e) {
    d.log.warn('could not set spellchecker languages', e)
  }

  ses.on('will-download', (_event, item) => {
    // Chromium writes *.crdownload and renames at the end, so two same-named downloads in a row
    // would both see the name as free. Reserve it until done. lstat so a dangling symlink isn't "free".
    const taken = (p: string) => pendingDownloads.has(p) || lstatExists(p)
    const target = uniquePath(d.downloads, item.getFilename(), taken)
    pendingDownloads.add(target)
    item.setSavePath(target)
    item.once('done', (_e, state) => {
      pendingDownloads.delete(target)
      if (state !== 'completed') {
        d.log.warn('download did not complete', { state })
        return
      }
      const n = new Notification({ title: 'Saved to Downloads', body: basename(target) })
      n.on('click', () => shell.showItemInFolder(target))
      n.show()
    })
  })
  return ses
}

export function accountForSession(ses: Session, accounts: Account[]): string | undefined {
  return accounts.find((a) => session.fromPartition(partitionOf(a.id)) === ses)?.id
```

```ts
// Sliding-window limiter: at most `max` calls per `windowMs` for each key.
export function createRateLimiter(max: number, windowMs: number, now: () => number = Date.now): (key: string | number) => boolean {
  const hits = new Map<string | number, number[]>()
  return (key) => {
    const t = now()
    const recent = (hits.get(key) ?? []).filter((h) => t - h < windowMs)
    const allowed = recent.length < max
    if (allowed) recent.push(t)
    hits.set(key, recent)
    return allowed
  }
}
```

```ts
      return
    case 'blocked':
```

```ts
const SIZE = 128

function normalize(img: NativeImage): NativeImage | null {
  if (img.isEmpty()) return null
  const { width, height } = img.getSize()
  const side = Math.min(width, height)
  return img
    .crop({ x: Math.floor((width - side) / 2), y: Math.floor((height - side) / 2), width: side, height: side })
    .resize({ width: SIZE, height: SIZE, quality: 'best' })
}

// <userData>/avatars/<accountId>.png, square 128 px. Returned as data URLs for the shell.
export class AvatarFiles {
  constructor(private readonly dir: string) {
    mkdirSync(dir, { recursive: true })
  }

  load(id: string): string | null {
    const file = this.file(id)
    if (!existsSync(file)) return null
    const img = nativeImage.createFromPath(file)
    return img.isEmpty() ? null : img.toDataURL()
  }

  save(id: string, img: NativeImage): string | null {
    const n = normalize(img)
    if (!n) return null
    writeFileSync(this.file(id), n.toPNG())
    return n.toDataURL()
  }

  remove(id: string): void {
    rmSync(this.file(id), { force: true })
  }

  private file(id: string): string {
    return join(this.dir, `${id}.png`)
  }
}
```

```ts
  const isWhatsAppOrigin = (url: string) => {
    try {
      return new URL(url).origin === WA_ORIGIN
    } catch {
      return false
    }
  }
  wc.on('did-start-navigation', (details) => {
    if (!details.isMainFrame || details.isSameDocument || isWhatsAppOrigin(details.url)) return
    wc.stop()
    d.log.warn('stopped a navigation away from WhatsApp', { accountId })
    performLink(classifyLink(details.url), wc, d.log)
  })
  wc.on('will-redirect', (details) => {
    if (!details.isMainFrame || isWhatsAppOrigin(details.url)) return
    details.preventDefault()
    d.log.warn('blocked a redirect away from WhatsApp', { accountId })
  })
  wc.on('will-navigate', (event, url) => {
    const action = classifyLink(url)
    if (action.kind === 'internal') return
    event.preventDefault()
    performLink(action, wc, d.log)
  })

  wc.on('context-menu', (_e, params) => popupContextMenu(wc, params, { dev: d.dev, log: d.log }))

  wc.on('before-input-event', (event, input) => {
    const action = matchShortcut(input)
    if (!action) return
    event.preventDefault()
    d.onShortcut(action)
  })

  wc.on('page-title-updated', (_e, title) => d.hub.setRuntime(accountId, { unread: parseUnread(title) }))

  wc.on('did-start-loading', () => {
    const state = d.hub.runtimeOf(accountId).state
    if (state !== 'offline' && state !== 'failed') d.hub.setRuntime(accountId, { state: 'loading' })
  })
  wc.on('did-stop-loading', () => {
    if (d.hub.runtimeOf(accountId).state === 'loading') d.hub.setRuntime(accountId, { state: 'ready' })
  })
  wc.on('did-fail-load', (_e, code, description, _url, isMainFrame) => {
    if (!isMainFrame) return
    const state = loadStateForError(code)
    if (!state) return
    d.hub.setRuntime(accountId, { state })
    d.log.warn('guest load failed', { accountId, code, description })
  })
  wc.on('did-finish-load', () => wc.setZoomLevel(d.hub.account(accountId)?.zoom ?? 0))

  wc.on('render-process-gone', (_e, details) => {
    if (details.reason === 'clean-exit') return
    d.log.error('guest renderer gone', { accountId, reason: details.reason })
    d.sendShell({ type: 'remount', id: accountId })
  })
  watchHang(wc, () => {
    d.log.warn('guest hung, remounting', { accountId })
    d.hub.setRuntime(accountId, { state: 'hung' })
    d.sendShell({ type: 'remount', id: accountId })
  })

  wc.once('destroyed', () => d.registry.detach(wc.id))
}
```

Retry after a failure is driven by the shell (the Retry button) and by the guest's `online`
event. `did-start-loading` therefore does not clear `offline`/`failed`; the next successful
`did-stop-loading` after a retry does, because retry sets the state back to `loading` first
(Task 20, `account:reload`).

```ts
// Native notifications for WhatsApp's own (intercepted by the guest shim). Clicking one opens
// the account, then replays WhatsApp's onclick in that guest, which opens the right chat.
export class Notifier {
  private live = new Map<string, Notification>()
  private byTag = new Map<string, string>() // `${accountId}:${tag}` → live key

  constructor(private readonly d: NotifierDeps) {}

  show(accountId: string, p: GuestPayload<'guest:notify'>): void {

    if (!account || !Notification.isSupported()) return
    const s = this.d.hub.settings().notifications
    const focused = this.d.isWindowFocused()
    if (!shouldNotify({ enabled: s.enabled, muted: account.muted, windowFocused: focused, activeId: this.d.hub.active(), accountId })) return

    if (!allowNotification(accountId)) {
      this.d.log.warn('notifications rate-limited', { accountId })
      return
    }

    const text = formatNotification({ accountName: account.name, title: p.title, body: p.body, showPreview: s.showPreview })
    const decoded = s.showPreview && p.icon ? nativeImage.createFromDataURL(p.icon) : undefined
    const icon = decoded && !decoded.isEmpty() ? decoded : undefined
    const n = new Notification({ title: text.title, body: text.body, silent: !s.sound || p.silent, icon })
    const key = `${accountId}:${p.nid}`
    if (p.tag) {
      const tagKey = `${accountId}:${p.tag}`
      const previous = this.byTag.get(tagKey)
      if (previous) {
        this.live.get(previous)?.close()
        this.live.delete(previous)
      }
      this.byTag.set(tagKey, key)
    }
    n.on('click', () => {
      this.live.delete(key)
      this.d.reveal(accountId)
      this.d.registry.guestOf(accountId)?.send('guest:notify-click', { nid: p.nid })
    })
    n.on('close', () => this.live.delete(key))
    this.live.get(key)?.close()
    this.live.set(key, n)
    n.show()
    if (s.markUrgent && !focused) this.d.flash()
  }

  close(accountId: string, nid: string): void {
    const key = `${accountId}:${nid}`
    this.live.get(key)?.close()
    this.live.delete(key)
  }
```

```ts
export function registerGuestIpc(d: { hub: StateHub; registry: Registry; notifier: Notifier; avatars: AvatarFiles; log: Logger }): void {

    ipcMain.on(channel, (event, raw) => {
      const accountId = d.registry.accountOf(event.sender.id)
      if (!accountId) return
      const parsed = guestChannels[channel].safeParse(raw)
      if (!parsed.success) {
        d.log.warn('invalid guest payload', { channel })

      fn(accountId, parsed.data as GuestPayload<C>, event.sender)

    if (!online) d.hub.setRuntime(id, { state: 'offline' })
    else if (state === 'offline' || state === 'failed') {
      d.hub.setRuntime(id, { state: 'loading' })
      sender.reload()

    if (image.isEmpty()) return

    if (url) d.hub.setRuntime(id, { avatar: url })
  })
  on('guest:notify', (id, payload) => d.notifier.show(id, payload))
  on('guest:notify-close', (id, { nid }) => d.notifier.close(id, nid))
}
```

```ts
// Runs in WhatsApp's MAIN world through contextBridge.executeInMainWorld, which serialises it
// with Function.toString(). So: no imports, no outer variables, no class fields (bundler helpers
// would be left behind). Everything it needs is declared inside.
export function installNotificationShim(): void {
  interface Bridge {
    show(payload: unknown): void
    close(nid: string): void
  }
  interface ShimNote extends EventTarget {
    nid: string
    title: string
    body: string
    tag: string
    icon: string
    silent: boolean
    data: unknown
    onclick: ((this: ShimNote, ev: Event) => unknown) | null
    onclose: ((this: ShimNote, ev: Event) => unknown) | null
    onshow: ((this: ShimNote, ev: Event) => unknown) | null
    onerror: ((this: ShimNote, ev: Event) => unknown) | null
  }
  const w = window as unknown as {
    __diwanNotify?: Bridge
    __diwanShimmed?: boolean
    __diwanClick?: (nid: string) => void
  }
  const bridge = w.__diwanNotify
  if (!bridge || w.__diwanShimmed) return
  w.__diwanShimmed = true

  const live = new Map<string, ShimNote>()
  let seq = 0

  // Re-encode to a small PNG inside the page: Electron's nativeImage can't decode WebP, and main
  // only accepts PNG/JPEG with capped dimensions. Gives '' when anything fails.
  const toPngDataUrl = (src: string, max: number): Promise<string> => {
    if (!src || typeof createImageBitmap !== 'function' || typeof OffscreenCanvas !== 'function') return Promise.resolve('')
    const work = fetch(src)
      .then((r) => r.blob())
      .then((blob) => createImageBitmap(blob))
      .then(async (bmp) => {
        const scale = Math.min(1, max / Math.max(bmp.width, bmp.height))
        const w = Math.max(1, Math.round(bmp.width * scale))
        const h = Math.max(1, Math.round(bmp.height * scale))
        const canvas = new OffscreenCanvas(w, h)
        canvas.getContext('2d')?.drawImage(bmp, 0, 0, w, h)
        bmp.close()
        const png = await canvas.convertToBlob({ type: 'image/png' })
        return new Promise<string>((resolve) => {
          const fr = new FileReader()
          fr.onload = () => resolve(String(fr.result))
          fr.onerror = () => resolve('')
          fr.readAsDataURL(png)
        })
      })
    const timeout = new Promise<string>((resolve) => setTimeout(() => resolve(''), 1500))
    return Promise.race([work, timeout]).catch(() => '')
  }

  class DiwanNotification extends EventTarget {
    constructor(title: string, options: NotificationOptions & { icon?: string } = {}) {
      super()
      const self = this as unknown as ShimNote
      self.title = String(title)
      self.body = String(options.body ?? '')
      self.tag = String(options.tag ?? '')
      self.icon = String(options.icon ?? '')
      self.silent = Boolean(options.silent)
      self.data = options.data ?? null
      self.onclick = null
      self.onclose = null
      self.onshow = null
      self.onerror = null
      seq += 1
      self.nid = `n${seq}`
      live.set(self.nid, self)
      void toPngDataUrl(self.icon, 256).then((icon) =>
        bridge?.show({
          nid: self.nid,
          title: self.title,
          body: self.body,
          tag: self.tag,
          icon: icon.startsWith('data:image/png;base64,') ? icon : '',
          silent: self.silent,
        }),
      )
      setTimeout(() => {
        const ev = new Event('show')
        self.onshow?.call(self, ev)
        self.dispatchEvent(ev)
      }, 0)
    }

    static get permission(): NotificationPermission {
      return 'granted'
    }

    static requestPermission(callback?: (p: NotificationPermission) => void): Promise<NotificationPermission> {
      callback?.('granted')
      return Promise.resolve('granted')
    }

    static get maxActions(): number {
      return 0
    }

    close(): void {
      const self = this as unknown as ShimNote
      if (!live.delete(self.nid)) return
      bridge?.close(self.nid)
      const ev = new Event('close')
      self.onclose?.call(self, ev)
      self.dispatchEvent(ev)
    }
  }

  w.__diwanClick = (nid: string) => {
    const n = live.get(nid)
    if (!n) return
    live.delete(nid)
    window.focus()
    const ev = new Event('click', { cancelable: true })
    n.onclick?.call(n, ev)
    n.dispatchEvent(ev)
  }

  Object.defineProperty(window, 'Notification', { value: DiwanNotification, writable: true, configurable: true })
}
```

```ts
// Best effort: find the user's own profile picture in WhatsApp Web. Own picture sits inside the
// left nav rail (new layout) or the top-left header (old layout); contact pictures don't.
export interface AvatarCandidate {
  top: number
  left: number
  width: number
  height: number
  alt: string
  round: boolean
}

export const MIN_SCORE = 250

export function scoreAvatar(c: AvatarCandidate): number {
  if (c.width < 15 || c.height < 15 || c.width > 100 || c.height > 100) return -1
  let score = 0
  if (c.left + c.width <= 80) score += 150
  if (c.top < 70) score += 200
  else if (c.top < 120) score += 50
  if (c.width >= 30 && c.width <= 50) score += 80
  else if (c.width >= 20 && c.width <= 60) score += 40
  if (c.alt) score += 50
  if (c.round) score += 60
  return score
}

export function pickAvatar(doc: Document): HTMLImageElement | null {
  let best: HTMLImageElement | null = null
  let bestScore = -1
  for (const img of Array.from(doc.querySelectorAll('img'))) {
    if (!img.src || img.src.startsWith('data:image/gif') || /emoji|status/.test(img.src)) continue
    const r = img.getBoundingClientRect()
    const radius = getComputedStyle(img).borderRadius
    const score = scoreAvatar({
      top: r.top,
      left: r.left,
      width: r.width,
      height: r.height,
      alt: img.alt,
      round: radius === '50%' || Number.parseInt(radius, 10) > 15,
    })
    if (score > bestScore) {
      best = img
      bestScore = score
    }
  }
  return best && bestScore >= MIN_SCORE ? best : null
}

// Re-encoded to a PNG of at most 256 px: main accepts only PNG/JPEG with capped dimensions,
// and Electron can't decode WebP.
export async function findAvatarDataUrl(doc: Document): Promise<string | null> {
  const img = pickAvatar(doc)
  if (!img) return null
  try {
    const bmp = await createImageBitmap(await (await fetch(img.src)).blob())
    const scale = Math.min(1, 256 / Math.max(bmp.width, bmp.height))
    const w = Math.max(1, Math.round(bmp.width * scale))
    const h = Math.max(1, Math.round(bmp.height * scale))
    const canvas = new OffscreenCanvas(w, h)
    canvas.getContext('2d')?.drawImage(bmp, 0, 0, w, h)
    bmp.close()
    const png = await canvas.convertToBlob({ type: 'image/png' })
    return await new Promise<string | null>((resolve) => {
      const fr = new FileReader()
      fr.onload = () => resolve(String(fr.result))
      fr.onerror = () => resolve(null)
      fr.readAsDataURL(png)
    })
  } catch {
    return null
  }
}
```

```ts
export function initials(name: string): string {
  const words = name.trim().split(/\s+/).filter(Boolean)
  if (words.length === 0) return '?'
  if (words.length === 1) return Array.from(words[0]).slice(0, 2).join('').toUpperCase()
  return (Array.from(words[0])[0] + Array.from(words[1])[0]).toUpperCase()
}

export function formatBytes(n: number): string {
  if (n < 1024) return `${n} B`
  const units = ['KB', 'MB', 'GB', 'TB']
  let v = n / 1024
  let i = 0
  while (v >= 1024 && i < units.length - 1) {
    v /= 1024
    i++
  }
  return `${v >= 10 ? Math.round(v) : Math.round(v * 10) / 10} ${units[i]}`
}

export function cleanError(e: unknown): string {
  const msg = e instanceof Error ? e.message : String(e)
  return msg.replace(/^Error invoking remote method '[^']+': (?:Error: )?/, '')
}
```

```ts
import { useEffect } from 'react'

export function useEscape(onEscape: () => void, enabled = true): void {
  useEffect(() => {
    if (!enabled) return
    const onKey = (e: KeyboardEvent) => {
      if (e.key === 'Escape') {
        e.stopPropagation()
        onEscape()
      }
    }
    window.addEventListener('keydown', onKey)
    return () => window.removeEventListener('keydown', onKey)
  }, [onEscape, enabled])
}
```

```tsx
export function Rail() {

  const openDialog = useUi((s) => s.openDialog)
  const openMenu = useUi((s) => s.openMenu)
  const openSettings = useUi((s) => s.openSettings)
  const [dragId, setDragId] = useState<string | null>(null)
  const [drop, setDrop] = useState<{ id: string; where: 'before' | 'after' } | null>(null)
  if (!snap) return null
  const { accounts, runtime, activeId } = snap

  const onDragOver = (id: string) => (e: DragEvent) => {
    if (!dragId) return
    e.preventDefault()
    const rect = (e.currentTarget as HTMLElement).getBoundingClientRect()
    setDrop({ id, where: e.clientY < rect.top + rect.height / 2 ? 'before' : 'after' })
  }
  const onDrop = () => {
    if (dragId && drop && dragId !== drop.id) {
      const ids = accounts.map((a) => a.id).filter((id) => id !== dragId)
      const at = ids.indexOf(drop.id) + (drop.where === 'after' ? 1 : 0)
      ids.splice(at, 0, dragId)
      void call('account:reorder', { ids })
    }
    setDragId(null)
    setDrop(null)
  }

  return (
    <nav aria-label="Accounts" className="drag flex w-[68px] shrink-0 flex-col items-center gap-2.5 border-r border-line bg-bg-1 pt-3.5 pb-3">

              openMenu({ id: a.id, x: e.clientX, y: e.clientY })

              e.dataTransfer.effectAllowed = 'move'
              setDragId(a.id)
            }}
            onDragOver={onDragOver(a.id)}
            onDrop={onDrop}
            onDragEnd={() => {
              setDragId(null)
              setDrop(null)
            }}
          />
        ))}
      </div>
```

```tsx
// One <webview> per account, all mounted at once and stacked. Inactive ones are hidden with
// visibility (not display:none) so they keep their size and switching never reflows WhatsApp.
export function GuestView({ id, active, mountKey }: { id: string; active: boolean; mountKey: number }) {
  const ref = useRef<HTMLWebViewElement>(null)
  useEffect(() => {
    if (active) ref.current?.focus()
  }, [active, mountKey])
  return (
    <webview
      key={mountKey}
      ref={ref}
      partition={`persist:acc-${id}`}
      src="https://web.whatsapp.com/"
      // @ts-expect-error Electron only checks that the attribute exists; React needs a string value.
      allowpopups="true"
      className={`absolute inset-0 h-full w-full ${active ? 'visible z-10' : 'invisible'}`}
    />
  )
}
```

```ts
import { BADGE_KEYS } from '../shared/unread'
import { MARK_PATHS } from './icons'

const SIZE = 32

// Draws the tray icon once per badge state (none, 1–9, 9+) and returns PNG data URLs for main.
export async function drawTrayBadges(): Promise<Record<string, string>> {
  await document.fonts.ready
  const out: Record<string, string> = {}
  for (const key of BADGE_KEYS) {
    const canvas = document.createElement('canvas')
    canvas.width = SIZE
    canvas.height = SIZE
    const ctx = canvas.getContext('2d')
    if (!ctx) continue
    ctx.fillStyle = '#18181b'
    ctx.beginPath()
    ctx.roundRect(1, 1, SIZE - 2, SIZE - 2, 8)
    ctx.fill()
    ctx.save()
    ctx.translate(4, 4)
    ctx.scale((SIZE - 8) / 24, (SIZE - 8) / 24)
    ctx.strokeStyle = '#fafafa'
    ctx.lineWidth = 2
    ctx.lineCap = 'round'
    ctx.lineJoin = 'round'
    for (const d of MARK_PATHS) ctx.stroke(new Path2D(d))
    ctx.restore()
    if (key !== 'none') {
      const r = key === '9+' ? 9 : 8
      ctx.fillStyle = '#3b82f6'
      ctx.beginPath()
      ctx.arc(SIZE - r, r, r, 0, Math.PI * 2)
      ctx.fill()
      ctx.fillStyle = '#ffffff'
      ctx.font = `700 ${key === '9+' ? 9 : 11}px Inter`
      ctx.textAlign = 'center'
      ctx.textBaseline = 'middle'
      ctx.fillText(key, SIZE - r, r + 0.5)
    }
    out[key] = canvas.toDataURL('image/png')
  }
  return out
}
```

```tsx
import { useEffect } from 'react'
import { call } from './api'
import { AccountMenu } from './components/AccountMenu'
import { Dialogs } from './components/Dialogs'
import { Rail } from './components/Rail'
import { RecoveryBanner } from './components/RecoveryBanner'
import { Stage } from './components/Stage'
import { WindowControls } from './components/WindowControls'
import { SettingsPage } from './settings/SettingsPage'
import { useUi } from './state'
import { drawTrayBadges } from './trayBadges'
import { Toasts } from './ui/Toasts'

export function App() {
  const snap = useUi((s) => s.snap)
  const settingsOpen = useUi((s) => s.settingsOpen)

  useEffect(() => {
    const { setSnap, bumpRemount, openSettings } = useUi.getState()
    void call('state:get').then((s) => s && setSnap(s))
    const offState = window.diwan.onState(setSnap)
    const offCommand = window.diwan.onCommand((c) => {
      if (c.type === 'remount') bumpRemount(c.id)
      else if (c.type === 'openSettings') openSettings()
    })
    void drawTrayBadges().then((images) => call('tray:setBadges', images))
    return () => {
      offState()
      offCommand()
    }
  }, [])

  // Offer the Nexus import once, on a first run that has no accounts yet.
  useEffect(() => {
    if (snap?.importAvailable && snap.accounts.length === 0 && !useUi.getState().dialog) {
      useUi.getState().openDialog({ kind: 'import' })
    }
  }, [snap?.importAvailable, snap?.accounts.length])

  return (
    <div className="flex h-full flex-col">
      {snap?.settings.showWindowControls && <WindowControls />}
      {snap?.recovery && <RecoveryBanner />}
      <div className="flex min-h-0 flex-1">
        <Rail />
        <Stage />
      </div>
      {settingsOpen && <SettingsPage />}
      <AccountMenu />
      <Dialogs />
      <Toasts />
    </div>
  )
}
```
